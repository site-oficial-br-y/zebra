const modAtaque = () => (1 + (S.form.a - 3) * 0.09) * S.estilo.atk;

function novoJogo(form, estilo){
  const modoEscolhido = document.querySelector('.sp[data-modo].on')?.dataset.modo || 'auto';
  const velEscolhida  = +(document.querySelector('.sp[data-sp].on')?.dataset.sp ?? 1);

}
const vagasDe = g => S.slots.filter(s=>s.grupo===g && !s.jog).length;
const completo = () => S.slots.every(s=>s.jog);

function bloqueio(p){
  if (S.usados.has(p.id)) return 'já escalado';
  if (!vagasDe(p.grupo)) return ROT[p.grupo] + ' cheio';
  return null;
}
const temEscolha = t => t.elenco.some(p=>!bloqueio(p));

function trocaTatica(form, estilo){

function rolar(){
  if (completo()) return;
  let livres = DB.times.filter(t=>!S.sorteados.has(t.id) && temEscolha(t));
  if (!livres.length){ S.sorteados.clear(); livres = DB.times.filter(temEscolha); }
  if (!livres.length){
    S.atual = null; pinta();
    $('roll').textContent = 'Sem jogador disponível'; $('roll').disabled = true;
    return;
  }
  S.atual = livres[Math.floor(Math.random()*livres.length)];
  S.sorteados.add(S.atual.id);
  const m = $('marca');
  if (m){ m.style.animation='none'; void m.offsetWidth; m.style.animation='tremeDado .45s ease-in-out'; }
  pinta();
}
function escala(p){
  if (bloqueio(p)) return;
  const s = S.slots.find(s=>s.grupo===p.grupo && !s.jog);
  s.jog = {...p, de:`${S.atual.clube} ${S.atual.ano}`};
  S.ultimo = S.slots.indexOf(s);
  S.usados.add(p.id);
  S.atual = null;
  pinta();
  if (completo()) revelar(); else rolar();
}

  const xi = S.slots.map(s=>s.jog);
  S.over = Math.round(xi.reduce((a,p)=>a+p.over,0)/xi.length);
  $('ovTat').textContent = S.form.n + ' · ' + S.estilo.n;
  $('xi').innerHTML = xi.map(p=>`<div class="c"><span class="ov">${p.over}</span><div class="nm">${p.nome}</div><div class="sm">${ROT[p.grupo]} · ${p.de}</div></div>`).join('');
  const pontos = S.slots.map(s=>({rot:s.rot, x:s.x, y:s.y, jog:s.jog}));
  desenhaCampo($('campoRev'), pontos, false);
  $('ovTeam').textContent = '··';
  mostra('scr-reveal');

  const rapido = S.vel === 0;
  const bolas = [...$('campoRev').querySelectorAll('.pos')];
  for (let i = 0; i < bolas.length; i++){
    if (!rapido) await sleep(105);
    bolas[i].querySelector('.bola b').textContent = pontos[i].jog.over;
    bolas[i].classList.add('novo');
  }
  if (!rapido) await sleep(260);
  const alvo = S.over;
  if (rapido){ $('ovTeam').textContent = alvo; return; }
  /* conta por tempo, nao por quadro: se a aba perder o foco o numero
     ainda chega no valor certo em vez de congelar no meio */
  const DUR = 620, t0 = performance.now();
  const sobe = agora => {
    const p = Math.min(1, (agora - t0) / DUR);
    $('ovTeam').textContent = Math.round(alvo * (1 - Math.pow(1 - p, 3)));
    if (p < 1) requestAnimationFrame(sobe); else $('ovTeam').textContent = alvo;
  };
  requestAnimationFrame(sobe);
  setTimeout(()=>{ $('ovTeam').textContent = alvo; }, DUR + 120);
}

/* ---------- simulação ---------- */
const poisson = l => { let k=0,p=1,L=Math.exp(-l); do{k++;p*=Math.random()}while(p>L); return k-1; };
const MEDIA_BASE = 82.5, ESTICA = 2.2;
const forca = time => {
  const m = time.elenco.reduce((s,p)=>s+p.over,0) / time.elenco.length;
  return MEDIA_BASE + (m - MEDIA_BASE) * ESTICA;
};
function marcador(xi){
  const peso = p => p.over * ({ATA:6,MEI:3,DEF:1,GOL:0}[p.grupo]);
  const tot = xi.reduce((s,p)=>s+peso(p),0);
  let r = Math.random()*tot;
  for (const p of xi){ r -= peso(p); if (r<=0) return p.nome; }
  return xi[xi.length-1].nome;
}
function simula(meuXi, meuOv, opp){
  const of = forca(opp);
  const lam = (dif,mod) => Math.min(8, 1.35*Math.exp(dif/8)*mod);
  let a = Math.min(9, poisson(lam(meuOv-of, modAtaque())));
  let b = Math.min(9, poisson(lam(of-meuOv, modDefesa())));
  const ev = [];
  const min = () => 1 + Math.floor(Math.random()*90);
  for (let i=0;i<a;i++) ev.push({m:min(), meu:true,  who:marcador(meuXi)});
  for (let i=0;i<b;i++) ev.push({m:min(), meu:false, who:marcador(opp.elenco)});
  ev.sort((x,y)=>x.m-y.m);
  return {a,b,ev,of};
}
function penaltis(meuOv, of){
  let a=0,b=0;
  for(let i=0;i<5;i++){ if(Math.random() < .70 + (meuOv-of)/200) a++; if(Math.random() < .70 + (of-meuOv)/200) b++; }
  while(a===b){ const x=Math.random()<.72, y=Math.random()<.72; if(x)a++; if(y)b++; }
  return [a,b];
}

const FASES = ['Grupos','Grupos','Grupos','Oitavas','Quartas','Semifinal','Final'];
const sleep = ms => new Promise(r=>setTimeout(r,ms));

async function campanha(){
  mostra('scr-camp');
  const xi = S.slots.map(s=>s.jog);
  const pool = DB.times.slice().sort(()=>Math.random()-.5).sort((x,y)=>forca(x)-forca(y));
  const escolhidos = [];
  for (let i=0;i<7;i++){
    const faixa = Math.floor(pool.length * (i+1)/8);
    const idx = Math.min(pool.length-1, faixa + Math.floor(Math.random()*3));
    escolhidos.push(pool.splice(idx,1)[0]);
  }
  const box = $('matches'); box.innerHTML = '';
  let saldoPro = 0, saldoContra = 0, eliminado = false, campeao = false, pontos = 0;

  for (let i=0;i<7;i++){
    const opp = escolhidos[i], fase = FASES[i], mata = i>=3;
    const el = document.createElement('div');
    el.className = 'match';
    el.innerHTML = `<div class="top">
        <div class="fase">${fase.toUpperCase()}</div>
        <div class="opp"><span class="flag" style="width:24px;height:17px;background:linear-gradient(105deg,${opp.cores[0]} 0 50%,${opp.cores[1]} 50% 100%)"></span>${opp.clube} ${opp.ano}</div>
        <div><span class="sc d">0 – 0</span> <span class="clock">0'</span></div>
      </div><div class="evs"></div>`;
    box.appendChild(el);
    const sc = el.querySelector('.sc'), ck = el.querySelector('.clock'), evs = el.querySelector('.evs');
    el.classList.add('vivo');
    const r = simula(xi, S.over, opp);
    let a=0,b=0,k=0;
    const stepAgora = () => S.vel===0 ? 0 : Math.round(55/S.vel);

    for (let m=1;m<=90;m++){
      while (k<r.ev.length && r.ev[k].m===m){
        const e = r.ev[k++];
        if (e.meu) a++; else b++;
        sc.textContent = `${a} – ${b}`;
        const d = document.createElement('div');
        d.className = 'ev' + (e.meu?'':' c');
        d.innerHTML = `<span class="m">${e.m}'</span><span class="who">${e.who}</span>`;
        evs.appendChild(d);
      }
      sc.className = 'sc ' + (a>b?'w':a<b?'l':'d');
      const st = stepAgora();
      ck.textContent = m + "'";
      if (st) await sleep(st);
    }
    ck.textContent = "90'"; ck.classList.add('fim');
    el.classList.remove('vivo');
    saldoPro += a; saldoContra += b;

    if (!mata){
      pontos += a>b ? 3 : a===b ? 1 : 0;
      if (i===2){
        const passou = pontos >= PTS_CLASSIFICA;
        const d = document.createElement('div');
        d.className = 'ev' + (passou?'':' c');
        d.innerHTML = `<span class="m">${pontos}pt</span><span class="who">${passou?'Classificado para as oitavas':'Eliminado na fase de grupos'}</span>`;
        evs.appendChild(d);
        if (!passou){ eliminado = 'fase de grupos'; break; }
      }
    }
    if (mata && a===b){
      const [pa,pb] = penaltis(S.over, r.of);
      const d = document.createElement('div');
      d.className = 'ev' + (pa>pb?'':' c');
      d.innerHTML = `<span class="m">PEN</span><span class="who">${pa} x ${pb} nos pênaltis</span>`;
      evs.appendChild(d);
      sc.className = 'sc ' + (pa>pb?'w':'l');
      if (pb>pa){ eliminado = fase; break; }
    } else if (mata && a<b){ eliminado = fase; break; }
    if (i===6 && !eliminado) campeao = true;
    if (i < 6 && !eliminado){
      if (S.modo === 'manual') await esperaProximo();
      else { const pausa = stepAgora(); if (pausa) await sleep(pausa*4); }
    }
  }

  const v = $('verdict'); v.hidden = false;
  let goleada = null;
  for (const el of box.querySelectorAll('.sc')){
    const [x,y] = el.textContent.split('–').map(n=>+n.trim());
    if (x > y && (!goleada || x-y > goleada.d || (x-y === goleada.d && x > goleada.x)))
      goleada = {x, y, d:x-y};
  }
  const onde = String(eliminado);
  $('vbig').textContent = campeao ? 'CAMPEÃO'
    : onde === 'fase de grupos' ? 'CAIU NOS GRUPOS' : 'CAIU NAS ' + onde.toUpperCase();
  $('vbig').className = 'big' + (campeao ? ' seven' : '');
  $('vtxt').innerHTML = `Time de <b>${S.over}</b> de overall, no <b>${S.form.n}</b> ${S.estilo.n.toLowerCase()}.<br>
    ${saldoPro} gols marcados, ${saldoContra} sofridos.
    ${goleada ? `<br>Maior goleada: <b style="color:var(--verde)">${goleada.x} a ${goleada.y}</b>.` : ''}`;
  v.scrollIntoView({behavior:'smooth'});
}
function esperaProximo(){

    const w = $('proxWrap'), b = $('prox');
    w.hidden = false;
    w.scrollIntoView({behavior:'smooth', block:'nearest'});
    b.onclick = () => { w.hidden = true; b.onclick = null; resolve(); };
  });
}
