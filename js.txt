  const sobe = agora => {
    const p = Math.min(1, (agora - t0) / DUR);
    $('ovTeam').textContent = Math.round(alvo * (1 - Math.pow(1 - p, 3)));
    if (p < 1) requestAnimationFrame(sobe); else $('ovTeam').textContent = alvo;
  };
  requestAnimationFrame(sobe);
  setTimeout(()=>{ $('ovTeam').textContent = alvo; }, DUR + 120);
}

/* ---------- simulação ---------- */
const poisson = l => { let k=0,p=1,L=Math.exp(-l); do{k++;p*=Math.random()}while(p>L); return k-1; };
const MEDIA_BASE = 82.5;
const forca = time => {
  const m = time.elenco.reduce((s,p)=>s+p.over,0) / time.elenco.length;
  return MEDIA_BASE + (m - MEDIA_BASE) * BASES[BASE].estica;
};
function marcador(xi){
  const peso = p => p.over * ({ATA:6,MEI:3,DEF:1,GOL:0}[p.grupo]);
  const tot = xi.reduce((s,p)=>s+peso(p),0);
  let r = Math.random()*tot;
  for (const p of xi){ r -= peso(p); if (r<=0) return p.nome; }
  return xi[xi.length-1].nome;
}
function simula(meuXi, meuOv, opp, mods, empatar){
  // o duelo local manda a força e os modificadores prontos, com os dois lados
  // medidos pela mesma régua; nos outros modos sai tudo do estado atual
  const of = opp.forca != null ? opp.forca : forca(opp);
  // dividir por 8 fazia a média de gols explodir: 15 pontos de diferença
  // já pediam 8 gols. Com 16 o placar fica na casa do futebol de verdade.
  const lam = (dif,mod) => Math.min(3.6, 1.35*Math.exp(dif/16)*mod);
  let a = Math.min(6, poisson(lam(meuOv-of, mods ? mods.atk : modAtaque())));
  let b = Math.min(6, poisson(lam(of-meuOv, mods ? mods.sof : modDefesa())));
  if (empatar) b = a;
  const ev = [];
  const min = () => 1 + Math.floor(Math.random()*90);
  for (let i=0;i<a;i++) ev.push({m:min(), meu:true,  who:marcador(meuXi)});
  for (let i=0;i<b;i++) ev.push({m:min(), meu:false, who:marcador(opp.elenco)});
  ev.sort((x,y)=>x.m-y.m);
  return {a,b,ev,of};
}
/* Quem bate pênalti: os melhores do time primeiro, goleiro por último,
   e depois volta ao começo da fila se a disputa esticar. */
const filaDePenalti = xi => xi.slice().sort((p,q) =>
  (p.grupo === 'GOL') - (q.grupo === 'GOL') || q.over - p.over);

function penaltis(meuXi, meuOv, opp, of){
  const filas = [filaDePenalti(meuXi), filaDePenalti(opp.elenco)];
  const forcas = [meuOv, of];
  const gols = [0, 0];
  const cobrancas = [];
  let n = 0;

  const bate = lado => {
    const quem = filas[lado][n % filas[lado].length];
    // quem bate melhor converte mais: 70% de base, mais a qualidade do
    // batedor e a diferença de nível entre os times
    const chance = .62 + (quem.over - 78) / 100 + (forcas[lado] - forcas[1-lado]) / 260;
    const fez = Math.random() < Math.max(.45, Math.min(.95, chance));
    if (fez) gols[lado]++;
    cobrancas.push({ meu: lado === 0, nome: quem.nome, fez });
    return fez;
  };

  for (n = 0; n < 5; n++){ bate(0); bate(1); }
  // alternadas: para na primeira rodada em que um faz e o outro perde
  while (gols[0] === gols[1] && n < 15){
    const x = bate(0), y = bate(1);
    if (x !== y) break;
    n++;
  }
  return { a: gols[0], b: gols[1], cobrancas };
}

const sleep = ms => new Promise(r=>setTimeout(r,ms));

/* ---------- uma partida: cria o cartão, anima e devolve o resultado ---------- */
/* Serve à campanha do modo clássico e às três competições da temporada. */
async function jogaPartida({ xi, opp, box, fase, mata, casa, mods, empatar, eu }){
  const el = document.createElement('div');
  el.className = 'match';
  el.innerHTML = `<div class="top">
      <div class="fase">${fase.toUpperCase()}</div>
      <div class="quemjoga"><span class="meu">${eu || 'Seu time'}</span>
        <span class="x">x</span>
        <span class="opp"><span class="flag" style="width:24px;height:17px;background:linear-gradient(105deg,${opp.cores[0]} 0 50%,${opp.cores[1]} 50% 100%)"></span>${opp.clube} ${opp.ano}</span></div>
      <div><span class="sc d">0 – 0</span> <span class="clock">0'</span></div>
    </div><div class="evs"></div>`;
  box.appendChild(el);
  el.scrollIntoView({ behavior:'smooth', block:'nearest' });
  const sc = el.querySelector('.sc'), ck = el.querySelector('.clock'), evs = el.querySelector('.evs');
  el.classList.add('vivo');

  const r = simula(xi, S.over, opp, mods, empatar);
  let a = 0, b = 0, k = 0;
  const stepAgora = () => S.vel === 0 ? 0 : Math.round(55 / S.vel);

  for (let m = 1; m <= 90; m++){
    while (k < r.ev.length && r.ev[k].m === m){
      const e = r.ev[k++];
      if (e.meu) a++; else b++;
      sc.textContent = `${a} – ${b}`;
      const d = document.createElement('div');
      d.className = 'ev' + (e.meu ? '' : ' c');
      d.innerHTML = `<span class="m">${e.m}'</span><span class="who">${e.who}</span>`;
      evs.appendChild(d);
    }
    sc.className = 'sc ' + (a>b ? 'w' : a<b ? 'l' : 'd');
    const st = stepAgora();
    ck.textContent = m + "'";
    if (st) await sleep(st);
  }
  ck.textContent = "90'"; ck.classList.add('fim');
  el.classList.remove('vivo');

  const saida = { a, b, of: r.of, el, evs, sc, venceu: a > b, empate: a === b, penaltis: null };

  if (mata && a === b){
    const pen = penaltis(xi, S.over, opp, r.of);
    pen.cobrancas.forEach((c, i) => {
      const d = document.createElement('div');

/* sorteia adversários, do mais fraco ao mais forte */
function sorteiaAdversarios(quantos, deFraco = true, base = null){
  // o bolo pode trazer o mesmo time várias vezes, para dar peso a um grupo;
  // tirar as repetições antes do sorteio anularia esse peso, então elas só
  // são descartadas na hora de fechar a chave
  const pool = (base || elencos()).slice()
    .sort(() => Math.random()-.5).sort((x,y) => forca(x)-forca(y));
  const fora = [], jaEstao = new Set();
  for (let i = 0; i < quantos && pool.length; i++){
    const faixa = Math.floor(pool.length * (i+1) / (quantos+1));
    let idx = Math.min(pool.length-1, faixa + Math.floor(Math.random()*3));
    let tentativas = 0;
    while (jaEstao.has(pool[idx].id) && tentativas++ < pool.length){
      idx = (idx + 1) % pool.length;
    }
    const t = pool.splice(idx, 1)[0];
    jaEstao.add(t.id);
    fora.push(t);
  }
  return deFraco ? fora : fora.reverse();
}

  }

async function partidaLocal(){
  const [xi1, xi2] = LOCAL.times;
  const media = xi => xi.reduce((s,p)=>s+p.over,0) / xi.length;
  /* os dois lados passam pelo mesmo esticador: no duelo ninguém pode
     ganhar força só por estar do lado do adversário na conta */
  const estica = o => MEDIA_BASE + (o - MEDIA_BASE) * BASES[BASE].estica;
  const f1 = estica(media(xi1)), f2 = estica(media(xi2));
  const m1 = modsDe(LOCAL.forms[0], LOCAL.estilos[0]);
  const m2 = modsDe(LOCAL.forms[1], LOCAL.estilos[1]);
  const adversario = { clube:'Jogador 2', ano:'', cores:['#e9c46a','#0a0d0a'],
                       elenco: xi2, forca: f2 };
  const serie = [0, 0];
  const historico = [];

  /* melhor de três, com os mesmos onze. Empate na partida vai para os
     pênaltis, então toda partida sai com um vencedor e a série sempre fecha. */
  for (let jogo = 1; jogo <= 3; jogo++){
    mostra('scr-camp');
    $('campTitulo').textContent = ATALHO_PEN
      ? 'Duelo local · teste de pênaltis' : 'Duelo local · melhor de 3';
    $('matches').innerHTML = '';
    $('proxWrap').hidden = true;
    $('verdict').hidden = true;

    const el = document.createElement('div');
    el.className = 'duelo';
    el.innerHTML = `<div class="lado${serie[0]>serie[1]?' g':''}">
        <div class="q">Jogador 1</div><div class="o">${LOCAL.overs[0]}</div>
        <div class="f">${LOCAL.forms[0].n} · ${LOCAL.estilos[0].n}</div></div>
      <div class="vs"><b>${serie[0]} – ${serie[1]}</b><span>${jogo}ª de 3</span></div>
      <div class="lado${serie[1]>serie[0]?' g':''}">
        <div class="q">Jogador 2</div><div class="o">${LOCAL.overs[1]}</div>
        <div class="f">${LOCAL.forms[1].n} · ${LOCAL.estilos[1].n}</div></div>`;
    $('matches').appendChild(el);

    const antes = S.over;
    S.over = f1;
    const r = await jogaPartida({ xi: xi1, opp: adversario, box: $('matches'),
                                  fase: jogo + 'ª partida', mata: false, eu: 'Jogador 1',
                                  mods: { atk: m1.atk * m2.sof, sof: m2.atk * m1.sof },
                                  empatar: ATALHO_PEN });
    S.over = antes;

    const pen = r.a === r.b ? await disputaPenaltisLocal(xi1, xi2) : null;
    const g1 = pen ? pen.g[0] : r.a, g2 = pen ? pen.g[1] : r.b;
    serie[g1 > g2 ? 0 : 1]++;
    historico.push({ jogo, a:r.a, b:r.b, pen });

    if (serie[0] === 2 || serie[1] === 2) break;
    await placarDaSerie(serie, jogo + 1);
  }
  fimDaSerie(serie, historico);
}

/* ---------- pênaltis na mão, só no duelo local ---------- */
const PEN = { RMAX:125, RMIN:6, RALVO:40, JANELA:23, CICLOS:5 };
/* o anel fecha e abre; quanto melhor o batedor, mais devagar e mais tempo
   ele tem na janela. O tempo manda na animação, não a contagem de quadros. */
const cicloDoBatedor = over => 950 + (over - 73) * 32;

function filaLocal(xi){
  return xi.slice().sort((p,q) => (p.grupo==='GOL') - (q.grupo==='GOL') || q.over - p.over);
}

function anelDoBatedor(over, direcao){
  return new Promise(resolve => {
    const palco = $('penPalco'), anel = $('penAnel');
    const ciclo = cicloDoBatedor(over);
    let bateu = false, raf = 0, fim = 0;
    const t0 = performance.now();
    const raio = t => PEN.RMIN + (PEN.RMAX - PEN.RMIN)
      * (Math.cos(2*Math.PI*((t - t0)/ciclo)) + 1) / 2;

    $('penVoltas').innerHTML = Array.from({length:PEN.CICLOS}, () => '<i class="volta"></i>').join('');
    /* o anel nasce no tamanho certo: se ficar em zero até o primeiro quadro,
       um toque imediato marca raio 0 e o chute sai por cima sem o jogador errar */
    anel.style.width = anel.style.height = (PEN.RMAX*2) + 'px';
    anel.classList.remove('perto');
    let voltaVista = -1;

    const quadro = agora => {
      if (bateu) return;
      const dt = agora - t0;
      if (dt > ciclo * PEN.CICLOS) return;
      const r = raio(agora);
      anel.style.width = anel.style.height = (r*2) + 'px';
      anel.classList.toggle('perto', Math.abs(r - PEN.RALVO) <= PEN.JANELA);
      const n = Math.floor(dt / ciclo);
      if (n !== voltaVista){
        voltaVista = n;
        [...$('penVoltas').children].forEach((el,i) =>
          el.className = 'volta' + (i < n ? ' usada' : i === n ? ' agora' : ''));
      }
      raf = requestAnimationFrame(quadro);
    };
    /* o anel nasce grande e fecha. Só o raio não diz se o toque foi cedo
       ou tarde: com o anel abrindo de volta, raio grande é atraso. Por isso
       sai junto a fase do ciclo. */
    /* travar é uma coisa, responder é outra: o chute tranca tudo no toque
       e só devolve o resultado quando a bola termina de voar. Se as duas
       coisas andassem juntas, o cronômetro das cinco voltas caía no meio
       do voo e dizia que o jogador não tinha batido. */
    const tranca = () => {
      bateu = true;
      cancelAnimationFrame(raf); clearTimeout(fim);
      palco.onpointerdown = null;
    };
    raf = requestAnimationFrame(quadro);
    fim = setTimeout(() => { if (!bateu){ tranca(); resolve(null); } },
                     ciclo * PEN.CICLOS + 60);
    palco.onpointerdown = e => {
      e.preventDefault();
      if (bateu) return;
      const t = performance.now(), r = raio(t);
      tranca();
      const f = document.createElement('div');
      f.className = 'flash';
      f.style.width = f.style.height = (r*2) + 'px';
      palco.appendChild(f);
      setTimeout(() => f.remove(), 460);
      chutaBola(direcao, Math.abs(r - PEN.RALVO) <= PEN.JANELA);
      setTimeout(() => resolve({ r, fechando: Math.sin(2*Math.PI*((t - t0)/ciclo)) > 0 }), 620);
    };
  });
}

async function disputaPenaltisLocal(xi1, xi2){
  const filas = [filaLocal(xi1), filaLocal(xi2)];
  const g = [0, 0], feitas = [];
  let n = 0;

  const escolha = await telaPassagem({
    titulo: 'Pênaltis', cima: 'Deu empate',
    aviso: 'Cinco cobranças para cada um. Vocês podem bater na mão, '
         + 'ou deixar a sorte resolver.',
    botao: 'Bater na mão', botao2: 'Disputa automática',
  });
  if (escolha === 2) return penaltisAutoLocal(filas, g, feitas);

  /* o batedor vai primeiro e o goleiro adivinha depois: é o goleiro que
     tem de chutar o canto, não o contrário */
  const cobra = async lado => {
    const quem = filas[lado][n % 11];
    const outro = lado === 0 ? 2 : 1;

    await telaPassagem({
      vez: lado + 1, cima: 'Pênaltis · cobrança',
      aviso: `Sua vez de bater com <b>${quem.nome}</b>. `
           + 'Escolha o canto e acerte o tempo.',
    });
    mostra('scr-pen'); pintaPlacarPen(g, feitas);
    $('penQuem').textContent = 'Jogador ' + (lado + 1) + ' · cobra';
    $('penQuem').className = 'tag pen-quem bate';
    const ladoBat = await escolheLado(quem.nome, 'Para onde você bate?', '');

    $('penTitulo').textContent = quem.nome;
    $('penSub').textContent = 'Toque quando o anel chegar na bola';
    $('penLados').hidden = true;
    $('penPalcoWrap').hidden = false;
    const t = await anelDoBatedor(quem.over, ladoBat);
    $('penPalcoWrap').hidden = true;

    await telaPassagem({
      vez: outro, cima: 'Pênaltis · defesa',
      aviso: 'A bola já saiu. Você é o goleiro: <b>chute o canto</b>.',
    });
    mostra('scr-pen'); pintaPlacarPen(g, feitas);
    $('penQuem').textContent = 'Jogador ' + outro + ' · defende';
    $('penQuem').className = 'tag pen-quem pega';
    const ladoGol = await escolheLado('Você defende', 'Para onde você se joga?', 'ouro');

    const dentro = t && Math.abs(t.r - PEN.RALVO) <= PEN.JANELA;
    const cedo = t && (t.fechando ? t.r > PEN.RALVO : t.r < PEN.RALVO);
    let fez, motivo;
    if (t === null){ fez = false; motivo = 'deixou o anel dar as cinco voltas'; }
    else if (!dentro){ fez = false;
      motivo = cedo ? 'cedo demais, mandou por cima' : 'tarde demais, chute fraco'; }
    else if (ladoGol === ladoBat){ fez = false; motivo = 'o goleiro foi no mesmo canto'; }
    else { fez = true; motivo = 'no canto, sem chance'; }

    // meio segundo de suspense antes de dizer no que deu
    $('penTitulo').textContent = '...';
    $('penSub').textContent = 'A bola está indo';
    $('penLados').hidden = true;
    await sleep(500);

    if (fez) g[lado]++;
    feitas.push({ lado, nome: quem.nome, fez });
    pintaPlacarPen(g, feitas);
    $('penTitulo').textContent = fez ? 'GOL' : 'PERDEU';
    $('penTitulo').className = 'pen-t ' + (fez ? 'g' : 'r');
    $('penSub').textContent = motivo;
    await sleep(1600);
    $('penTitulo').className = 'pen-t';
  };

  for (n = 0; n < 5; n++){
    await cobra(0);
    if (g[0] - g[1] > 4 - n) break;
    await cobra(1);
    if (g[1] - g[0] > 4 - n) break;
  }
  while (g[0] === g[1]){
    await cobra(0); await cobra(1);
    n++;
  }
  return { g, feitas };
}

/* Pontos corridos: joga as rodadas e devolve a colocação.
   Os outros dezenove times pontuam sozinhos, com força parecida com a real. */
async function disputaBrasileirao(xi){
  cabecalhoTemp('Brasileirão', TEMP.brasileirao + ' rodadas');
  const advs = sorteiaAdversarios(TEMP.brasileirao);
  let pontos = 0, gp = 0, gc = 0;
  for (let i = 0; i < advs.length; i++){
    const r = await jogaPartida({ xi, opp: advs[i], box: $('matches'),
                                  fase: (i+1) + 'ª rodada', mata: false });
    pontos += r.venceu ? 3 : r.empate ? 1 : 0;
    gp += r.a; gc += r.b;
    const d = document.createElement('div');
    d.className = 'ev';
    d.innerHTML = `<span class="m">${pontos}pt</span><span class="who">${pontos} ponto${pontos===1?'':'s'} em ${i+1} jogo${i?'s':''}</span>`;
    r.evs.appendChild(d);
    if (i < advs.length - 1) await entreJogos();
  }
  // Os outros do campeonato pontuam conforme a própria força: antes era
  // sorteio puro e a Chapecoense podia terminar na frente do Flamengo.
  const media = elencos().reduce((s,t) => s + forca(t), 0) / elencos().length;
  const tabela = elencos()
    .filter(t => t !== advs[0])
    .slice(0, 19)
    .map(t => {
      const chance = Math.max(.10, Math.min(.55, .30 + (forca(t) - media) / 40));
      let p = 0;
      for (let r = 0; r < TEMP.brasileirao; r++){
        const x = Math.random();
        p += x < chance ? 3 : x < chance + .27 ? 1 : 0;
      }
      return { nome: t.clube, pts: p, cores: t.cores };
    });
  tabela.push({ nome: 'Seu time', pts: pontos, eu: true, cores: ['#7BE495','#0a0d0a'] });
  tabela.sort((a,b) => b.pts - a.pts || (a.eu ? -1 : b.eu ? 1 : 0));
  const posicao = tabela.findIndex(t => t.eu) + 1;
  desenhaTabela(tabela, posicao);
  return { pontos, posicao, gp, gc };
}

function desenhaTabela(tabela, posicao){
  const zona = i => i === 0 ? 'campeao' : i < TEMP.vagaLiberta ? 'liberta'
    : i >= tabela.length - 4 ? 'z4' : '';
  const el = document.createElement('div');
  el.className = 'tabela-wrap';
  el.innerHTML = `<div class="tabela-cab"><span>Classificação final</span>
      <span class="sua">${posicao}º lugar</span></div>
    <table class="tabela"><tbody>` + tabela.map((t,i) => `
      <tr class="${t.eu ? 'eu' : ''} ${zona(i)}">
        <td class="p">${i+1}</td>
        <td class="c"><span class="flag" style="width:18px;height:13px;background:linear-gradient(105deg,${t.cores[0]} 0 50%,${t.cores[1]} 50% 100%)"></span>${t.nome}</td>
        <td class="pt">${t.pts}</td>
      </tr>`).join('') + '</tbody></table>';
  $('matches').appendChild(el);
  el.scrollIntoView({ behavior:'smooth', block:'nearest' });
}

async function disputaMataMata(xi, nome, quantos, faseNomes, comCabecalho = true, base = null){
  if (comCabecalho) cabecalhoTemp(nome, quantos + ' jogos');
  const advs = sorteiaAdversarios(quantos, false, base);
  for (let i = 0; i < quantos; i++){
    const r = await jogaPartida({ xi, opp: advs[i], box: $('matches'),
                                  fase: faseNomes[i], mata: true });
    if (!r.venceu) return { campeao: false, caiu: faseNomes[i] };
    if (i < quantos - 1) await entreJogos();
  }
  return { campeao: true, caiu: null };
}

async function disputaLibertadores(xi){
  cabecalhoTemp('Libertadores', 'grupos e mata-mata');
  const grupo = sorteiaAdversarios(TEMP.libGrupos, true, CONTINENTE());
  let pts = 0;
  for (let i = 0; i < grupo.length; i++){
    const r = await jogaPartida({ xi, opp: grupo[i], box: $('matches'),
                                  fase: 'Grupo · jogo ' + (i+1), mata: false });
    pts += r.venceu ? 3 : r.empate ? 1 : 0;
    if (i === grupo.length - 1){
      const passou = pts >= PTS_CLASSIFICA;
      const d = document.createElement('div');
      d.className = 'ev' + (passou ? '' : ' c');
      d.innerHTML = `<span class="m">${pts}pt</span><span class="who">${passou ? 'Classificado' : 'Eliminado na fase de grupos'}</span>`;
      r.evs.appendChild(d);
      if (!passou) return { campeao: false, caiu: 'fase de grupos' };
    }
    await entreJogos();
  }
  return await disputaMataMata(xi, 'Libertadores', TEMP.libMata, FASES_MATA, false, CONTINENTE());
}

/* O campeão da América encara o melhor elenco da história do jogo.

async function temporada(){
  mostra('scr-camp');
  $('campTitulo').textContent = 'A temporada';
  const xi = S.slots.map(s => s.jog);
  $('matches').innerHTML = '';
  const tacas = [];

  await esperaCompeticao('Brasileirão', 'A temporada começa');
  const br = await disputaBrasileirao(xi);
  if (br.posicao === 1) tacas.push('Brasileirão');

  const temVaga = br.posicao <= TEMP.vagaLiberta;
  avisoVaga(br.posicao, temVaga);

  await esperaCompeticao('Copa do Brasil', 'Próxima competição');
  const copa = await disputaMataMata(xi, 'Copa do Brasil', TEMP.copa, FASES_MATA);
  if (copa.campeao) tacas.push('Copa do Brasil');

  let libertou = false;
  if (temVaga){
    await esperaCompeticao('Libertadores', 'Você tem vaga');
    const lib = await disputaLibertadores(xi);
    if (lib.campeao){ tacas.push('Libertadores'); libertou = true; }
  }

  if (libertou){
    await esperaCompeticao('Mundial', 'Campeão da América');
    const mun = await disputaMundial(xi);
    if (mun.campeao) tacas.push('Mundial');
  }

  fimDaTemporada(br, tacas);
}

/* Conta quantas vezes cada chave aparece e devolve as mais frequentes. */
function maisVezes(itens, quantos){
  const c = new Map();
  itens.forEach(k => c.set(k, (c.get(k)||0) + 1));
  return [...c.entries()].sort((a,b)=> b[1]-a[1] || a[0].localeCompare(b[0])).slice(0, quantos);
}
const clubeDe = de => String(de||'').replace(/\s+\d{4}$/, '') || '—';

function listaTop(itens, legenda){
  return '<ol class="top">' + itens.map(([k,q],i)=>
    `<li><span class="n">${i+1}</span>
       <span class="q">${k}</span>
       <span class="x">${q}${legenda||''}</span></li>`).join('') + '</ol>';
}

/* O melhor time: o campeão de maior overall. Sem título, o de maior overall. */
function melhorPartida(ps){
  const campeas = ps.filter(p=>p.campeao);
  const pool = campeas.length ? campeas : ps;
  return pool.reduce((a,b)=> b.overall > a.overall ? b : a, pool[0]);
}

function campoDaPartida(p){

      <div class="area t"></div><div class="area b"></div>
      <div class="linhas">${campo.innerHTML}</div>
    </div>`;
}

      <span class="subtag">Postura</span>
      ${listaTop(maisVezes(ps.map(p=>p.estilo), 3), 'x')}
    </div>
  </div>`;
}

function pintaConta(){
  $('deslogado').hidden = !!EU;
  $('logado').hidden = !EU;
  if (EU){
    $('meuApelido').textContent = EU.apelido || '—';
    $('perfilCorpo').innerHTML = '<p class="vazio">Carregando suas partidas…</p>';
    window.ZB.minhaLinha().then(meu=>{
      $('meuTit').textContent = meu ? meu.titulos : 0;
      $('minhasTacas').innerHTML = pintaTacas(meu);
      const taxa = meu && meu.partidas ? Math.round(meu.titulos / meu.partidas * 100) : 0;
      $('minhasStats').innerHTML = meu && meu.partidas
        ? `<span>Partidas <b>${meu.partidas}</b></span><span>Aproveitamento <b>${taxa}%</b></span><span>Melhor overall <b>${meu.melhor_overall}</b></span><span>Saldo <b>${meu.saldo>0?'+':''}${meu.saldo}</b></span><span>Maior goleada <b>${meu.maior_goleada}</b></span>`
        : '<span>Nenhuma partida registrada ainda</span>';
    }).catch(e => {
      // engolir o erro aqui escondia justamente o que eu precisava ver
      $('minhasStats').innerHTML =
        '<span style="color:var(--vermelho)">Não deu para ler suas estatísticas: '
        + (e.message || e) + '</span>';
    });
    window.ZB.minhasPartidas()
      .then(pintaPerfil)
      .catch(()=> $('perfilCorpo').innerHTML =
        '<p class="vazio">Não deu para carregar suas partidas agora. Tente de novo daqui a pouco.</p>');
  } else {
    const criando = modoConta === 'criar', recuperando = modoConta === 'recuperar';
    $('abasConta').hidden = recuperando;
    $('dicaRecuperar').hidden = !recuperando;
    $('tabEntrar').classList.toggle('on', modoConta === 'entrar');
    $('tabCriar').classList.toggle('on', criando);
    $('fApelido').hidden = !criando;
    $('fSenha').hidden = recuperando;
    $('aceiteBox').hidden = !criando;
    $('eloRecuperar').hidden = recuperando;
    $('eloVoltarLogin').hidden = !recuperando;
    $('fEnviar').textContent = recuperando ? 'Enviar link' : criando ? 'Criar conta' : 'Entrar';
    $('fSenha').autocomplete = criando ? 'new-password' : 'current-password';
  }
}

$('fEnviar').onclick = async () => {
  const email = $('fEmail').value.trim(), senha = $('fSenha').value;
  const apelido = $('fApelido').value.trim();
  if (!email) return diz('contaAviso','Preencha o e-mail.');
  if (modoConta === 'recuperar'){
    $('fEnviar').disabled = true; diz('contaAviso','');
    try { await window.ZB.recuperarSenha(email); } catch {}
    // Mesma resposta exista ou não a conta, de propósito.
    diz('contaAviso','Se existir conta com esse e-mail, o link de recuperação já está a caminho. Confira o spam.', true);
    $('fEnviar').disabled = false;
    return;
  }
  if (!senha) return diz('contaAviso','Preencha a senha.');
  if (modoConta === 'criar'){
    if (apelido.length < 2) return diz('contaAviso','Escolha um apelido de 2 a 18 letras.');
    if (!$('fAceite').checked) return diz('contaAviso','Marque que aceita os termos de uso e a política de privacidade.');
  }
  $('fEnviar').disabled = true; diz('contaAviso','');
  try {
    if (modoConta === 'criar') await window.ZB.criar(email, senha, apelido);
    else await window.ZB.entrar(email, senha);
    await carregaEu();
    diz('contaAviso', 'Tudo certo, ' + (EU?.apelido || '') + '.', true);
    pintaConta();
  } catch(e){ diz('contaAviso', e.message); }
  $('fEnviar').disabled = false;
};

/* ---- o campo da home inclina atrás do cursor (só onde existe mouse) ---- */
(function(){
  const campo = $('campo3d'), hero = document.querySelector('.hero');
  if (!campo || !hero) return;
  if (!matchMedia('(hover:hover) and (pointer:fine)').matches) return;
  if (matchMedia('(prefers-reduced-motion:reduce)').matches) return;

  let alvoX = 0, alvoY = 0, pedido = 0;
  const aplica = () => {
    pedido = 0;
    campo.style.setProperty('--ry', alvoX.toFixed(2) + 'deg');
    campo.style.setProperty('--rx', alvoY.toFixed(2) + 'deg');
  };
  hero.addEventListener('pointermove', e => {
    const r = hero.getBoundingClientRect();
    // -1 a 1 a partir do centro do bloco
    alvoX = ((e.clientX - r.left) / r.width  - .5) *  16;
    alvoY = ((e.clientY - r.top)  / r.height - .5) * -11;
    campo.classList.add('seguindo');
    // um ajuste por quadro: pointermove dispara muito mais que isso
    if (!pedido) pedido = requestAnimationFrame(aplica);
  });
  hero.addEventListener('pointerleave', () => {
    alvoX = alvoY = 0;
    campo.classList.remove('seguindo');
    if (!pedido) pedido = requestAnimationFrame(aplica);
  });
})();
</script>
