  S.ultimo = S.slots.indexOf(s);
  S.usados.add(p.id);
  S.atual = null;
  pinta();
  if (completo()) revelar(); else rolar();
}

/* ---------- simulação ---------- */
const poisson = l => { let k=0,p=1,L=Math.exp(-l); do{k++;p*=Math.random()}while(p>L); return k-1; };
const MEDIA_BASE = 82.5, ESTICA = 2.2;
const forca = time => {
  const m = time.elenco.reduce((s,p)=>s+p.over,0) / time.elenco.length;
  return MEDIA_BASE + (m - MEDIA_BASE) * ESTICA;
};
function marcador(xi){
  const peso = p => p.over * ({ATA:6,MEI:3,DEF:1,GOL:0}[p.grupo]);
  const tot = xi.reduce((s,p)=>s+peso(p),0);
  let r = Math.random()*tot;
  for (const p of xi){ r -= peso(p); if (r<=0) return p.nome; }
  return xi[xi.length-1].nome;
}
function simula(meuXi, meuOv, opp){
  const of = forca(opp);
  const lam = (dif,mod) => Math.min(8, 1.35*Math.exp(dif/8)*mod);
  let a = Math.min(9, poisson(lam(meuOv-of, modAtaque())));
  let b = Math.min(9, poisson(lam(of-meuOv, modDefesa())));
  const ev = [];
  const min = () => 1 + Math.floor(Math.random()*90);
  for (let i=0;i<a;i++) ev.push({m:min(), meu:true,  who:marcador(meuXi)});
  for (let i=0;i<b;i++) ev.push({m:min(), meu:false, who:marcador(opp.elenco)});
  ev.sort((x,y)=>x.m-y.m);
  return {a,b,ev,of};
}
function penaltis(meuOv, of){
  let a=0,b=0;
  for(let i=0;i<5;i++){ if(Math.random() < .70 + (meuOv-of)/200) a++; if(Math.random() < .70 + (of-meuOv)/200) b++; }
  while(a===b){ const x=Math.random()<.72, y=Math.random()<.72; if(x)a++; if(y)b++; }
  return [a,b];
}

const FASES = ['Grupos','Grupos','Grupos','Oitavas','Quartas','Semifinal','Final'];
const sleep = ms => new Promise(r=>setTimeout(r,ms));

async function campanha(){
  mostra('scr-camp');
  const xi = S.slots.map(s=>s.jog);
  const pool = DB.times.slice().sort(()=>Math.random()-.5).sort((x,y)=>forca(x)-forca(y));
  const escolhidos = [];
  for (let i=0;i<7;i++){
    const faixa = Math.floor(pool.length * (i+1)/8);
    const idx = Math.min(pool.length-1, faixa + Math.floor(Math.random()*3));
    escolhidos.push(pool.splice(idx,1)[0]);
  }
  const box = $('matches'); box.innerHTML = '';
  let saldoPro = 0, saldoContra = 0, eliminado = false, campeao = false, pontos = 0;

  for (let i=0;i<7;i++){
    const opp = escolhidos[i], fase = FASES[i], mata = i>=3;
    const el = document.createElement('div');
    el.className = 'match';
    el.innerHTML = `<div class="top">
        <div class="fase">${fase.toUpperCase()}</div>
        <div class="opp"><span class="flag" style="width:24px;height:17px;background:linear-gradient(105deg,${opp.cores[0]} 0 50%,${opp.cores[1]} 50% 100%)"></span>${opp.clube} ${opp.ano}</div>
        <div><span class="sc d">0 – 0</span> <span class="clock">0'</span></div>
      </div><div class="evs"></div>`;
    box.appendChild(el);
    const sc = el.querySelector('.sc'), ck = el.querySelector('.clock'), evs = el.querySelector('.evs');
    el.classList.add('vivo');
    const r = simula(xi, S.over, opp);
    let a=0,b=0,k=0;
    const stepAgora = () => S.vel===0 ? 0 : Math.round(55/S.vel);

    for (let m=1;m<=90;m++){
      while (k<r.ev.length && r.ev[k].m===m){
        const e = r.ev[k++];
        if (e.meu) a++; else b++;
        sc.textContent = `${a} – ${b}`;
        const d = document.createElement('div');
        d.className = 'ev' + (e.meu?'':' c');
        d.innerHTML = `<span class="m">${e.m}'</span><span class="who">${e.who}</span>`;
        evs.appendChild(d);
      }
      sc.className = 'sc ' + (a>b?'w':a<b?'l':'d');
      const st = stepAgora();
      ck.textContent = m + "'";
      if (st) await sleep(st);
    }
    ck.textContent = "90'"; ck.classList.add('fim');
    el.classList.remove('vivo');
    saldoPro += a; saldoContra += b;

    if (!mata){
      pontos += a>b ? 3 : a===b ? 1 : 0;
      if (i===2){
        const passou = pontos >= PTS_CLASSIFICA;
        const d = document.createElement('div');
        d.className = 'ev' + (passou?'':' c');
        d.innerHTML = `<span class="m">${pontos}pt</span><span class="who">${passou?'Classificado para as oitavas':'Eliminado na fase de grupos'}</span>`;
        evs.appendChild(d);
        if (!passou){ eliminado = 'fase de grupos'; break; }
      }
    }
    if (mata && a===b){
      const [pa,pb] = penaltis(S.over, r.of);
      const d = document.createElement('div');
      d.className = 'ev' + (pa>pb?'':' c');
      d.innerHTML = `<span class="m">PEN</span><span class="who">${pa} x ${pb} nos pênaltis</span>`;
      evs.appendChild(d);
      sc.className = 'sc ' + (pa>pb?'w':'l');
      if (pb>pa){ eliminado = fase; break; }
    } else if (mata && a<b){ eliminado = fase; break; }
    if (i===6 && !eliminado) campeao = true;
    if (i < 6 && !eliminado){
      if (S.modo === 'manual') await esperaProximo();
      else { const pausa = stepAgora(); if (pausa) await sleep(pausa*4); }
    }
  }

  const v = $('verdict'); v.hidden = false;
  let goleada = null;
  for (const el of box.querySelectorAll('.sc')){
    const [x,y] = el.textContent.split('–').map(n=>+n.trim());
    if (x > y && (!goleada || x-y > goleada.d || (x-y === goleada.d && x > goleada.x)))
      goleada = {x, y, d:x-y};
  }
  const onde = String(eliminado);
  $('vbig').textContent = campeao ? 'CAMPEÃO'
    : onde === 'fase de grupos' ? 'CAIU NOS GRUPOS' : 'CAIU NAS ' + onde.toUpperCase();
  $('vbig').className = 'big' + (campeao ? ' seven' : '');
  $('vtxt').innerHTML = `Time de <b>${S.over}</b> de overall, no <b>${S.form.n}</b> ${S.estilo.n.toLowerCase()}.<br>
    ${saldoPro} gols marcados, ${saldoContra} sofridos.
    ${goleada ? `<br>Maior goleada: <b style="color:var(--verde)">${goleada.x} a ${goleada.y}</b>.` : ''}`;
  v.scrollIntoView({behavior:'smooth'});

  if (window.ZB && EU){
    const salvou = await window.ZB.salvarPartida({
      overall: S.over, formacao: S.form.n, estilo: S.estilo.n,
      campeao, eliminado_em: campeao ? null : String(eliminado),
      gols_pro: saldoPro, gols_contra: saldoContra,
      maior_goleada: goleada ? goleada.d : 0,
      escalacao: xi.map(p=>({nome:p.nome, over:p.over, pos:ROT[p.grupo], de:p.de})),
    });
    if (salvou) $('vtxt').innerHTML += '<br><span style="color:var(--verde)">Partida registrada no ranking.</span>';
  } else if (window.ZB) {
    $('vtxt').innerHTML += '<br><span style="color:var(--apagado)">Entre na sua conta para valer no ranking.</span>';
  }
}

async function carregaEu(){

/* Conta quantas vezes cada chave aparece e devolve as mais frequentes. */
function maisVezes(itens, quantos){
  const c = new Map();
  itens.forEach(k => c.set(k, (c.get(k)||0) + 1));
  return [...c.entries()].sort((a,b)=> b[1]-a[1] || a[0].localeCompare(b[0])).slice(0, quantos);
}
const clubeDe = de => String(de||'').replace(/\s+\d{4}$/, '') || '—';

function listaTop(itens, legenda){
  return '<ol class="top">' + itens.map(([k,q],i)=>
    `<li><span class="n">${i+1}</span>
       <span class="q">${k}</span>
       <span class="x">${q}${legenda||''}</span></li>`).join('') + '</ol>';
}

/* O melhor time: o campeão de maior overall. Sem título, o de maior overall. */
function melhorPartida(ps){
  const campeas = ps.filter(p=>p.campeao);
  const pool = campeas.length ? campeas : ps;
  return pool.reduce((a,b)=> b.overall > a.overall ? b : a, pool[0]);
}

function campoDaPartida(p){
  const form = FORMACOES.find(f=>f.n === p.formacao);
  const xi = p.escalacao || [];
  if (!form || xi.length !== form.pos.length) return '';
  // A escalação foi gravada na ordem dos slots, então o índice reconstrói a posição.
  const pontos = form.pos.map((c,i)=> ({rot:c.rot, x:c.x, y:c.y, jog:xi[i]}));
  const campo = document.createElement('div');
  desenhaCampo(campo, pontos, true);
  return `<div class="campo mini perfil">
      <div class="marcacao"></div><div class="meio"></div><div class="circulo"></div>
      <div class="area t"></div><div class="area b"></div>
      <div class="linhas">${campo.innerHTML}</div>
    </div>`;
}

const ORDEM_FASES = ['fase de grupos','Oitavas','Quartas','Semifinal','Final'];
const NOME_FASE = { 'fase de grupos':'Grupos' };

function pintaPerfil(ps){
  const alvo = $('perfilCorpo');
  if (!ps.length){
    alvo.innerHTML = `<p class="vazio">Nenhuma partida registrada ainda.
      Monte um time, jogue a campanha até o fim e ela aparece aqui — com o seu melhor
      time, os jogadores que mais escala e até onde você costuma chegar.</p>`;
    return;
  }

  const todos = ps.flatMap(p => p.escalacao || []);
  const melhor = melhorPartida(ps);
  const fases = maisVezes(ps.map(p => p.campeao ? 'Campeão' : (p.eliminado_em || 'Grupos')));
  const linhas = ['Campeão', ...ORDEM_FASES.slice().reverse()]
    .map(f => [f, (fases.find(x=>x[0]===f) || [,0])[1]])
    .filter(([,q]) => q > 0);
  const maior = Math.max(...linhas.map(([,q])=>q), 1);

  alvo.innerHTML = `
  <div class="bloco">
    <div class="cab">
      <span class="tag">Seu melhor time</span>
      <span class="sub"><b>${melhor.overall}</b> de overall · ${melhor.formacao}
        ${melhor.estilo.toLowerCase()} · ${melhor.campeao ? 'campeão' : 'caiu ' + (melhor.eliminado_em === 'fase de grupos' ? 'nos grupos' : 'nas ' + melhor.eliminado_em.toLowerCase())}</span>
    </div>
    <div class="melhor">
      ${campoDaPartida(melhor)}
      <div class="xi" style="margin-top:0">${(melhor.escalacao||[]).map(j=>
        `<div class="c"><span class="ov">${j.over}</span><div class="nm">${j.nome}</div>
           <div class="sm">${j.pos} · ${j.de}</div></div>`).join('')}</div>
    </div>
  </div>

  <div class="duas">
    <div class="bloco">
      <span class="tag">Quem mais joga por você</span>
      ${listaTop(maisVezes(todos.map(j=>j.nome), 5), 'x')}
    </div>
    <div class="bloco">
      <span class="tag">De onde vem o seu time</span>
      ${listaTop(maisVezes(todos.map(j=>clubeDe(j.de)), 5), '')}
    </div>
  </div>

  <div class="duas">
    <div class="bloco">
      <span class="tag">Até onde você chega</span>
      <div class="barras">${linhas.map(([f,q])=>
        `<div class="barra${f==='Campeão'?' ouro':''}">
           <span class="et">${NOME_FASE[f] || f}</span>
           <span class="tr"><span class="fi" style="width:${Math.round(q/maior*100)}%"></span></span>
           <span class="qt">${q}</span>
         </div>`).join('')}</div>
    </div>
    <div class="bloco">
      <span class="tag">Como você joga</span>
      <span class="subtag">Formação</span>
      ${listaTop(maisVezes(ps.map(p=>p.formacao), 3), 'x')}
      <span class="subtag">Postura</span>
      ${listaTop(maisVezes(ps.map(p=>p.estilo), 3), 'x')}
    </div>
  </div>`;
}

function pintaConta(){
  $('deslogado').hidden = !!EU;
  $('logado').hidden = !EU;
  if (EU){
    $('meuApelido').textContent = EU.apelido || '—';
    $('perfilCorpo').innerHTML = '<p class="vazio">Carregando suas partidas…</p>';
    window.ZB.ranking().then(r=>{
      const meu = r.find(x=>x.user_id === EU.id);
      $('meuTit').textContent = meu ? meu.titulos : 0;
      const taxa = meu && meu.partidas ? Math.round(meu.titulos / meu.partidas * 100) : 0;
      $('minhasStats').innerHTML = meu && meu.partidas
        ? `<span>Partidas <b>${meu.partidas}</b></span><span>Aproveitamento <b>${taxa}%</b></span><span>Melhor overall <b>${meu.melhor_overall}</b></span><span>Saldo <b>${meu.saldo>0?'+':''}${meu.saldo}</b></span><span>Maior goleada <b>${meu.maior_goleada}</b></span>`
        : '<span>Nenhuma partida registrada ainda</span>';
    }).catch(()=>{});
    window.ZB.minhasPartidas()
      .then(pintaPerfil)
      .catch(()=> $('perfilCorpo').innerHTML =
        '<p class="vazio">Não deu para carregar suas partidas agora. Tente de novo daqui a pouco.</p>');
  } else {
    const criando = modoConta === 'criar', recuperando = modoConta === 'recuperar';
    $('abasConta').hidden = recuperando;
    $('dicaRecuperar').hidden = !recuperando;
    $('tabEntrar').classList.toggle('on', modoConta === 'entrar');
    $('tabCriar').classList.toggle('on', criando);
    $('fApelido').hidden = !criando;
    $('fSenha').hidden = recuperando;
    $('aceiteBox').hidden = !criando;
    $('eloRecuperar').hidden = recuperando;
    $('eloVoltarLogin').hidden = !recuperando;
    $('fEnviar').textContent = recuperando ? 'Enviar link' : criando ? 'Criar conta' : 'Entrar';
    $('fSenha').autocomplete = criando ? 'new-password' : 'current-password';
  }
}

$('fEnviar').onclick = async () => {
  const email = $('fEmail').value.trim(), senha = $('fSenha').value;
  const apelido = $('fApelido').value.trim();
  if (!email) return diz('contaAviso','Preencha o e-mail.');
  if (modoConta === 'recuperar'){
    $('fEnviar').disabled = true; diz('contaAviso','');
    try { await window.ZB.recuperarSenha(email); } catch {}
    // Mesma resposta exista ou não a conta, de propósito.
    diz('contaAviso','Se existir conta com esse e-mail, o link de recuperação já está a caminho. Confira o spam.', true);
    $('fEnviar').disabled = false;
    return;
  }
  if (!senha) return diz('contaAviso','Preencha a senha.');
  if (modoConta === 'criar'){
    if (apelido.length < 2) return diz('contaAviso','Escolha um apelido de 2 a 18 letras.');
    if (!$('fAceite').checked) return diz('contaAviso','Marque que aceita os termos de uso e a política de privacidade.');
  }
  $('fEnviar').disabled = true; diz('contaAviso','');
  try {
    if (modoConta === 'criar') await window.ZB.criar(email, senha, apelido);
    else await window.ZB.entrar(email, senha);
    await carregaEu();
    diz('contaAviso', 'Tudo certo, ' + (EU?.apelido || '') + '.', true);
    pintaConta();
  } catch(e){ diz('contaAviso', e.message); }
  $('fEnviar').disabled = false;
};
